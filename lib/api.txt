const API_BASE_URL =
  process.env.NEXT_PUBLIC_API_URL || "http://localhost:5000";

async function requestApi(path: string, options?: RequestInit) {
  const response = await fetch(`${API_BASE_URL}${path}`, {
    ...options,
    headers: {
      "Content-Type": "application/json",
      ...(options?.headers || {}),
    },
  });

  const data = await response.json().catch(() => null);

  if (!response.ok) {
    throw new Error(data?.message || "API request failed");
  }

  return data?.data ?? data;
}

export async function getSkillGap() {
  const data = await requestApi("/api/readiness/skill-gap");

  const list = Array.isArray(data) ? data : data?.skills || data?.skillGaps || [];

  return list.map((item: any) => ({
    skill: item.skill || item.skill_name || item.skillName,
    current: item.current || item.current_score || item.currentScore,
    required: item.required || item.required_score || item.requiredScore,
    demand: item.demand,
    priority: item.priority,
  }));
}

export async function getMarketDemand() {
  const data = await requestApi("/api/readiness/market-demand");

  const list = Array.isArray(data)
    ? data
    : data?.marketDemands || data?.demands || [];

  return list.map((item: any) => ({
    rank: item.rank || item.rank_order || item.rankOrder,
    skill: item.skill || item.skill_name || item.skillName,
    jobs: item.jobs || item.jobs_count || item.jobsCount,
    trend: item.trend || item.trend_score || item.trendScore,
    barWidth:
      item.barWidth ||
      item.bar_width ||
      item.trend_score ||
      item.trendScore ||
      60,
  }));
}

export async function submitOnboarding(payload: any) {
  return requestApi("/api/onboarding", {
    method: "POST",
    body: JSON.stringify(payload),
  });
}
